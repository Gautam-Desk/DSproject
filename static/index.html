Python Integration & Standalone TensorFlow Script

Easily integrate the trained TensorFlow model into your Python data science pipelines and notebooks.

Python Inference Snippet (Standalone) 
import json
import numpy as np
import tensorflow as tf
from tensorflow import keras

# 1. Load Vocabulary and Initialize Vectorizer
with open("models/vocab.json", "r", encoding="utf-8") as f:
    vocab = json.load(f)

vectorizer = keras.layers.TextVectorization(
    max_tokens=8000,
    output_mode='int',
    output_sequence_length=160,
    standardize='lower_and_strip_punctuation',
    vocabulary=vocab
)

# 2. Load Trained Keras Model
model = keras.models.load_model("models/best_fake_news_model.keras")

# 3. Predict News Authenticity
def predict_news(headline: str, body: str):
    full_text = f"{headline} - {body}"
    seq = vectorizer(tf.constant([full_text])).numpy()
    fake_prob = float(model.predict(seq, verbose=0)[0][0])
    
    return {
        "verdict": "FAKE" if fake_prob >= 0.5 else "REAL",
        "fake_probability": round(fake_prob * 100, 2),
        "real_probability": round((1.0 - fake_prob) * 100, 2)
    }

# Example Usage
sample_headline = "NASA James Webb Space Telescope Discovers Ancient Galaxy"
sample_body = "Astronomers observed luminous star formations 350M years after Big Bang."

result = predict_news(sample_headline, sample_body)
print("Classification Result:", result)
